# 系统级 C++ 设计面试题 03：操作系统与系统编程（11–15）

> 平台：Linux/POSIX + C++17。系统调用可能被信号中断；以下均为可讨论、可扩展的最小核心实现，而不是覆盖安全、日志轮转、权限和守护进程运维的完整产品。

## 11. Asynchronous Logger（P1）

**题目**：多线程 log(text) 快速入队，专用后台线程负责输出；限定队列容量，队列满时显式 drop，关闭时 flush 已接收消息；统计 drop 数。假定输出流寿命覆盖 logger。

~~~cpp
#include <atomic>
#include <condition_variable>
#include <deque>
#include <mutex>
#include <ostream>
#include <string>
#include <thread>

class AsyncLogger {
    std::mutex m_;
    std::mutex outputMutex_;
    std::condition_variable cv_, drained_;
    std::deque<std::string> q_;
    std::thread worker_;
    std::ostream& out_;
    size_t cap_;
    bool closed_ = false, writing_ = false;
    std::atomic<uint64_t> drops_{0};
    void run() {
        for (;;) {
            std::string line;
            {
                std::unique_lock<std::mutex> lk(m_);
                cv_.wait(lk, [&] { return closed_ || !q_.empty(); });
                if (q_.empty() && closed_) break;
                line = std::move(q_.front());
                q_.pop_front();
                writing_ = true;
            }
            try {
                std::lock_guard<std::mutex> outlk(outputMutex_);
                out_ << line << '\n';
            } catch (...) { /* production: report I/O failure */ }
            {
                std::lock_guard<std::mutex> lk(m_);
                writing_ = false;
                if (q_.empty()) drained_.notify_all();
            }
        }
        std::lock_guard<std::mutex> outlk(outputMutex_);
        out_.flush();
    }
public:
    AsyncLogger(std::ostream& out, size_t cap) : out_(out), cap_(cap) {
        if (!cap_) throw std::invalid_argument("capacity");
        worker_ = std::thread([this] { run(); });
    }
    bool log(std::string s) {
        std::lock_guard<std::mutex> lk(m_);
        if (closed_) return false;
        if (q_.size() >= cap_) { drops_.fetch_add(1); return false; }
        q_.push_back(std::move(s));
        cv_.notify_one();
        return true;
    }
    void flush() {
        std::unique_lock<std::mutex> lk(m_);
        drained_.wait(lk, [&] { return q_.empty() && !writing_; });
        lk.unlock();
        std::lock_guard<std::mutex> outlk(outputMutex_);
        out_.flush(); // serialized against worker output
    }
    uint64_t dropped() const { return drops_.load(); }
    void shutdown() {
        {
            std::lock_guard<std::mutex> lk(m_);
            closed_ = true;
        }
        cv_.notify_all();
        if (worker_.joinable()) worker_.join();
    }
    ~AsyncLogger() { shutdown(); }
};
~~~

**复杂度**：log 摊销 O(length)，队列内存 O(capacity×average message size)。**面试要点**：drop/block/overwrite 策略；日志排序语义；batching；tail latency；落盘同步 vs ostream flush；输出失败状态；flush 与 worker 对 ostream 的访问须串行化（此例用 outputMutex_，仅保证队列 drain 与 ostream flush，不保证磁盘 durability）。

## 12. Event-Driven I/O with epoll（P1）

**题目**：用 epoll ET 模式监听多 socket 的可读事件，必须读到 EAGAIN。允许注册/删除 fd，处理对端关闭。用 Linux-only 代码。

~~~cpp
#include <sys/epoll.h>
#include <unistd.h>
#include <fcntl.h>
#include <cerrno>
#include <stdexcept>
#include <array>

class EpollReadLoop {
    int ep_ = ::epoll_create1(EPOLL_CLOEXEC);
public:
    EpollReadLoop() {
        if (ep_ < 0) throw std::runtime_error("epoll_create1");
    }
    ~EpollReadLoop() { ::close(ep_); }
    void add(int fd) {
        int flags = ::fcntl(fd, F_GETFL, 0);
        if (flags < 0 || ::fcntl(fd, F_SETFL, flags | O_NONBLOCK) < 0)
            throw std::runtime_error("fcntl");
        epoll_event ev{};
        ev.events = EPOLLIN | EPOLLET | EPOLLRDHUP;
        ev.data.fd = fd;
        if (::epoll_ctl(ep_, EPOLL_CTL_ADD, fd, &ev) < 0)
            throw std::runtime_error("epoll_ctl");
    }
    void remove(int fd) {
        ::epoll_ctl(ep_, EPOLL_CTL_DEL, fd, nullptr);
        ::close(fd);
    }
    template<class OnBytes>
    void once(OnBytes onBytes, int timeoutMs = 100) {
        std::array<epoll_event, 64> events{};
        int n = ::epoll_wait(ep_, events.data(), events.size(), timeoutMs);
        if (n < 0) {
            if (errno == EINTR) return;
            throw std::runtime_error("epoll_wait");
        }
        for (int i = 0; i < n; ++i) {
            int fd = events[i].data.fd;
            bool closeFd = (events[i].events & (EPOLLERR | EPOLLHUP));
            for (;;) {
                char buf[4096];
                ssize_t r = ::read(fd, buf, sizeof(buf));
                if (r > 0) { onBytes(fd, buf, static_cast<size_t>(r)); continue; }
                if (r == 0) { closeFd = true; break; }
                if (errno == EINTR) continue;
                if (errno == EAGAIN || errno == EWOULDBLOCK) break;
                closeFd = true; break;
            }
            if (closeFd) remove(fd);
        }
    }
};
~~~

**复杂度**：事件驱动 O(ready fds + bytes)，注册 O(1) 摊销内核成本。**面试要点**：ET/LT；ET 不读到 EAGAIN 会卡住；partial read/write；EPOLLOUT 不能长期无条件注册；fd reuse/stale events，实际服务器应使用 generation + stable connection token，不能只用 fd 作为身份；回调不可在自身关闭 fd 后继续使用它。

## 13. Shared-Memory IPC / SPSC（P1）

**题目**：两个 Linux 进程共享固定记录 ring buffer；单 writer、单 reader；共享段由独立初始化步骤先建好；每次传 64-bit value。用 shm_open/ftruncate/mmap；读写用原子同步。

~~~cpp
#include <atomic>
#include <cstdint>
#include <sys/mman.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <stdexcept>

struct alignas(64) SharedCounterRing {
    static constexpr uint64_t CAP = 1024;
    alignas(64) std::atomic<uint64_t> head{0};
    alignas(64) std::atomic<uint64_t> tail{0};
    alignas(64) uint64_t slots[CAP]{};

    bool push(uint64_t value) {
        auto t = tail.load(std::memory_order_relaxed);
        auto h = head.load(std::memory_order_acquire);
        if (t - h == CAP) return false;
        slots[t % CAP] = value;
        tail.store(t + 1, std::memory_order_release);
        return true;
    }
    bool pop(uint64_t& value) {
        auto h = head.load(std::memory_order_relaxed);
        auto t = tail.load(std::memory_order_acquire);
        if (t == h) return false;
        value = slots[h % CAP];
        head.store(h + 1, std::memory_order_release);
        return true;
    }
};
// Run once before consumers attach; filename begins with '/'.
inline void initSharedRing(const char* name) {
    int fd = ::shm_open(name, O_CREAT | O_EXCL | O_RDWR, 0600);
    if (fd < 0) throw std::runtime_error("shm_open");
    if (::ftruncate(fd, sizeof(SharedCounterRing)) != 0) {
        ::close(fd); throw std::runtime_error("ftruncate");
    }
    void* p = ::mmap(nullptr, sizeof(SharedCounterRing),
                     PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    ::close(fd);
    if (p == MAP_FAILED) throw std::runtime_error("mmap");
    new (p) SharedCounterRing();
    ::munmap(p, sizeof(SharedCounterRing));
}
// Attach with shm_open(name, O_RDWR), mmap(MAP_SHARED), then use ring->push/pop.
// Each process unmaps on exit; owner calls shm_unlink when all peers are done.
~~~

**复杂度**：O(1) 每条记录；空间 O(CAP)。**面试要点**：跨进程 C++ 原子语义依赖平台/ABI，须验证 lock-free、CPU 可见性；初始化握手（上例不允许初始化与 attach 并发）；crash 后 head/tail 恢复；共享指针不可直接跨进程使用，改用 offset；huge pages、cache line 对齐、NUMA；buffer overflow/backpressure、stale generation。

## 14. Process Supervisor（P2）

**题目**：启动 child process，退出后按退避策略重启；父进程收到停止请求可先 TERM 后 KILL。下面实现简化的单进程 supervisor 循环，stop flag 由外部发出，execv 使用绝对路径。

~~~cpp
#include <atomic>
#include <chrono>
#include <csignal>
#include <stdexcept>
#include <sys/wait.h>
#include <unistd.h>
#include <thread>
#include <algorithm>
#include <cerrno>

class ProcessSupervisor {
    std::atomic<bool> stop_{false};
public:
    void request_stop() noexcept { stop_.store(true); }
    void run(const char* path, char* const argv[]) {
        int backoffMs = 100;
        while (!stop_.load()) {
            pid_t pid = ::fork();
            if (pid < 0) throw std::runtime_error("fork");
            if (pid == 0) { ::execv(path, argv); _exit(127); }
            int status = 0;
            bool stoppingChild = false;
            while (!stop_.load()) {
                pid_t r = ::waitpid(pid, &status, WNOHANG);
                if (r == pid) break;
                if (r < 0 && errno == EINTR) continue;
                if (r < 0) throw std::runtime_error("waitpid");
                std::this_thread::sleep_for(std::chrono::milliseconds(50));
            }
            if (stop_.load()) {
                if (::kill(pid, SIGTERM) == 0) stoppingChild = true;
                for (int i = 0; i < 20; ++i) {
                    if (::waitpid(pid, &status, WNOHANG) == pid) {
                        stoppingChild = false; break;
                    }
                    std::this_thread::sleep_for(std::chrono::milliseconds(50));
                }
                if (stoppingChild) {
                    ::kill(pid, SIGKILL);
                    while (::waitpid(pid, &status, 0) < 0 && errno == EINTR) {}
                }
                break;
            }
            std::this_thread::sleep_for(std::chrono::milliseconds(backoffMs));
            backoffMs = std::min(backoffMs * 2, 5000);
        }
    }
};
~~~

**复杂度**：监控 O(1) 周期轮询；**面试要点**：上例有外部信号 race：child 恰好已结束、PID 被重用时 kill 风险，生产版用 pidfd；子进程群清理、zombie、TERM timeout、crash loop、stable uptime 重置 backoff、环境变量、身份权限、健康检测。安全的跨进程监控不只靠 waitpid。

## 15. RAII Resource Manager for FDs（P1）

**题目**：给 sockets/files/epoll fd 统一所有权语义；禁止复制、支持移动，确保异常或 early return 自动释放。调用方不能同时拥有同一 fd。

~~~cpp
#include <unistd.h>
#include <utility>

class UniqueFd {
    int fd_ = -1;
public:
    explicit UniqueFd(int fd = -1) noexcept : fd_(fd) {}
    ~UniqueFd() { reset(); }
    UniqueFd(const UniqueFd&) = delete;
    UniqueFd& operator=(const UniqueFd&) = delete;
    UniqueFd(UniqueFd&& rhs) noexcept : fd_(rhs.release()) {}
    UniqueFd& operator=(UniqueFd&& rhs) noexcept {
        if (this != &rhs) reset(rhs.release());
        return *this;
    }
    int get() const noexcept { return fd_; }
    explicit operator bool() const noexcept { return fd_ >= 0; }
    int release() noexcept { return std::exchange(fd_, -1); }
    void reset(int next = -1) noexcept {
        int old = std::exchange(fd_, next);
        if (old >= 0) ::close(old); // never retry close on Linux EINTR
    }
};
~~~

**复杂度**：O(1)，空间 O(1)。**面试要点**：copy vs move ownership、fd=0 有效、release 与 reset、close EINTR 不重试、close 后 fd number 可重用、非阻塞网络 write/read 的错误处理、CLOEXEC 防止 exec 泄漏；多个线程使用 fd 时，RAII 管理对象析构前要停止使用者。
